cmake_minimum_required(VERSION 3.21)
project(iidx_bgm_bridge VERSION 0.1.0 LANGUAGES C CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 with MSVC is required")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}")
endforeach()
add_library(minhook STATIC vendor/minhook/src/buffer.c vendor/minhook/src/hook.c
  vendor/minhook/src/trampoline.c vendor/minhook/src/hde/hde64.c)
add_library(bgm_bridge SHARED src/plugin.cpp)
target_include_directories(bgm_bridge PRIVATE include)
target_link_libraries(bgm_bridge PRIVATE minhook)
add_executable(bgm-tests tests/test_core.cpp)
target_include_directories(bgm-tests PRIVATE include)
target_link_libraries(bgm-tests PRIVATE minhook)
add_executable(audio_codec tools/audio_codec.cpp)
add_executable(host-image-tests tests/test_host_image.cpp)
target_link_libraries(host-image-tests PRIVATE minhook)
target_link_libraries(audio_codec PRIVATE msacm32)
add_executable(loudness_meter tools/loudness_meter.cpp)
foreach(target bgm_bridge bgm-tests host-image-tests audio_codec loudness_meter)
  target_compile_options(${target} PRIVATE /W4 /WX /utf-8 /EHsc)
endforeach()
enable_testing()
add_test(NAME playback-and-detours COMMAND bgm-tests)
set_tests_properties(playback-and-detours PROPERTIES WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}")
